\section{まとめ}

\begin{frame}{まとめ}
  容量制約のないメトリック施設配置問題に対して、\\局所探索法に基づく近似アルゴリズムを紹介した。

  \begin{theorem}
    施設配置問題の局所最適解 \(\localopt\) の総コストは、高々 \(3\OPT\) である。
  \end{theorem}

  \begin{theorem}[\cite{charikar2005improved}]
    施設配置問題に対して，任意の定数 \(\varepsilon > 0\) に対し，
    % \alert{\(\tilde{\mathcal{O}}(n^2/\varepsilon)\)}
    入力と \(1/\varepsilon\) の多項式時間で \alert{\(1 + \sqrt{2} + \varepsilon\)} 倍の近似解を求めるスキームが存在する．
  \end{theorem}

\end{frame}

\section{Appendix}

\begin{frame}{コストの離散化テクニック}

  \begin{lemma}[\cite{korte2008combinatorial}]
    任意のメトリック施設配置問題のインスタンス \(\mathcal{I}\)、\(\gamma_{\max} \ge 1\) および \(0 < \varepsilon \le 1\) に対して、
    \(\OPT(I) = 0\) であるかどうかを判定でき、そうでない場合には \(\mathcal{O}(|\FacilitySet||\DemandSet|\log{|\FacilitySet||\DemandSet|})\) 時間で別のインスタンス \(\mathcal{I}'\) を構成できる。
    このとき、すべての施設コストおよび割り当てコストは
    \[
      \biggr\{ 0, 1, \ldots, \left\lceil \frac{2\gamma_{\max}(k + |\DemandSet|)^3}{\varepsilon} \right\rceil \biggl\}
    \]
    に属する整数であり、さらに任意の \(1 \le \gamma \le \gamma_{\max}\) に対して、
    \(I'\) においてコストが高々 \(\gamma \OPT(I')\) である解は、元のインスタンス \(I\) においてコストが高々 \((1 + \varepsilon)\gamma \OPT(I)\) である解に対応する。
  \end{lemma}
\end{frame}
