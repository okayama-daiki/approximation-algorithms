\section*{記法／変数のまとめ}

\begin{enumerate}
  \leftskip 2em

  \item[\(F\): ] 施設集合, Facility set
  \item[\(D\): ] 利用者集合, Demand set
  \item[\(f_i\): ] 施設 \(i\) の開設コスト
  \item[\(c(i,j)\): ] 施設 \(i\) と利用者 \(j\) 間の距離
  \item[\(S (\subseteq F)\): ] 開設する施設
  \item[\(\sigma: D \to S\): ] 利用者から施設への割当関数
\end{enumerate}
